<template>
  <div class="blur-container">
    <div class="control-panel">
      <div class="control-group">
        <label for="bg-url">Adres URL obrazu tła:</label>
        <input
          id="bg-url"
          type="text"
          v-model="backgroundImageUrl"
          @change="updateBackgroundImage"
          placeholder="Wklej link do obrazka..."
          style="background: #333; color: white; border: 1px solid #555; padding: 6px; border-radius: 4px; font-size: 12px;"
        />
      </div>

      <div class="control-group">
        <label for="opacity-range">Przezroczystość efektu:</label>
        <input
          id="opacity-range"
          type="range"
          min="0"
          max="1"
          step="0.1"
          v-model.number="opacity"
        />
      </div>

      <div class="status-display">
        Status: <span :class="{ 'text-ready': isReady }">{{ statusMessage }}</span>
      </div>
    </div>

    <div class="view-container">
      <video
        ref="videoRef"
        autoplay
        playsinline
        muted
        style="display: none;"
      ></video>

      <div
        ref="canvasContainerRef"
        class="webgl-canvas-container"
        :style="{ opacity: opacity }"
      ></div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, onBeforeUnmount } from 'vue';
import * as THREE from 'three';
import { FilesetResolver, ImageSegmenter } from '@mediapipe/tasks-vision';

// --- STAN REAKTYWNY VUE ---
const backgroundImageUrl = ref('https://images.unsplash.com/photo-1497366216548-37526070297c?auto=format&fit=crop&w=1280&q=80');
const opacity = ref(1.0);
const statusMessage = ref('Uruchamianie kamery i pobieranie AI...');
const isReady = ref(false);

// Referencje do elementów DOM
const videoRef = ref<HTMLVideoElement | null>(null);
const canvasContainerRef = ref<HTMLDivElement | null>(null);

// --- ZMIENNE POMOCNICZE (POZA REAKTYWNOŚCIĄ DLA WYDAJNOŚCI) ---
let imageSegmenter: ImageSegmenter | null = null;
let animationFrameId: number | null = null;
let lastVideoTime = -1;

// Obiekty Three.js
let scene: THREE.Scene;
let camera: THREE.OrthographicCamera;
let renderer: THREE.WebGLRenderer;
let shaderMaterial: THREE.ShaderMaterial;
let webcamTexture: THREE.VideoTexture;
let maskTexture: THREE.Texture;
let backgroundTexture: THREE.Texture;
const textureLoader = new THREE.TextureLoader();

// Bufor Canvas 2D do konwersji maski z MediaPipe na teksturę Three.js
let maskCanvas: HTMLCanvasElement;
let maskCtx: CanvasRenderingContext2D | null;

// --- KOD SHADERÓW (GLSL) ---
const vertexShader = `
  varying vec2 vUv;
  void main() {
    vUv = uv;
    // Odbicie lustrzane obrazu z kamery dla naturalnego efektu
    vUv.x = 1.0 - vUv.x;
    gl_Position = vec4(position, 1.0);
  }
`;

const fragmentShader = `
  varying vec2 vUv;
  uniform sampler2D tWebcam;
  uniform sampler2D tMask;
  uniform sampler2D tBackground;

  void main() {
    vec4 cameraColor = texture2D(tWebcam, vUv);

    // Pobieramy tło (odwracamy UV z powrotem, aby obrazek tła nie był lustrzany)
    vec2 bgUv = vec2(1.0 - vUv.x, vUv.y);
    vec4 backgroundColor = texture2D(tBackground, bgUv);

    // Próbkowanie maski
    vec4 maskColor = texture2D(tMask, vUv);

    // NAPRAWA KRAWDĘDZI: Pobieramy bezpośrednią, płynną wartość przejścia (wygładzanie liniowe).
    // Karta graficzna automatycznie przefiltruje krawędzie, eliminując pikselowe schodki.
    float isPerson = maskColor.r;

    // Płynne zmiksowanie postaci i sztucznego tła
    gl_FragColor = mix(cameraColor, backgroundColor, isPerson);
  }
`;

// --- INICJALIZACJA THREE.JS ---
const initThree = (container: HTMLDivElement, video: HTMLVideoElement) => {
  scene = new THREE.Scene();
  camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1);

  renderer = new THREE.WebGLRenderer({ antialias: true, alpha: false });
  renderer.setSize(container.clientWidth, container.clientHeight);
  container.appendChild(renderer.domElement);

  // Rejestracja tekstury kamery
  webcamTexture = new THREE.VideoTexture(video);
  webcamTexture.minFilter = THREE.LinearFilter;
  webcamTexture.magFilter = THREE.LinearFilter;

  // Ładowanie obrazu tła
  backgroundTexture = textureLoader.load(backgroundImageUrl.value);
  backgroundTexture.minFilter = THREE.LinearFilter;
  backgroundTexture.magFilter = THREE.LinearFilter;

  // Tworzenie płótna buforowego pod maskę
  maskCanvas = document.createElement('canvas');
  maskCanvas.width = video.videoWidth || 640;
  maskCanvas.height = video.videoHeight || 480;
  maskCtx = maskCanvas.getContext('2d');

  // Powiązanie bufora z teksturą (LinearFilter jest kluczowy dla gładkich brzegów)
  maskTexture = new THREE.CanvasTexture(maskCanvas);
  maskTexture.minFilter = THREE.LinearFilter;
  maskTexture.magFilter = THREE.LinearFilter;

  // Przygotowanie płaszczyzny ekranowej
  const geometry = new THREE.PlaneGeometry(2, 2);
  shaderMaterial = new THREE.ShaderMaterial({
    vertexShader,
    fragmentShader,
    uniforms: {
      tWebcam: { value: webcamTexture },
      tMask: { value: maskTexture },
      tBackground: { value: backgroundTexture }
    },
    depthWrite: false,
    depthTest: false
  });

  const mesh = new THREE.Mesh(geometry, shaderMaterial);
  scene.add(mesh);
};

// Zmiana tła w locie
const updateBackgroundImage = () => {
  if (shaderMaterial && backgroundImageUrl.value) {
    textureLoader.load(backgroundImageUrl.value, (texture) => {
      texture.minFilter = THREE.LinearFilter;
      texture.magFilter = THREE.LinearFilter;
      shaderMaterial.uniforms.tBackground.value = texture;
      backgroundTexture.dispose();
      backgroundTexture = texture;
    });
  }
};

// --- URUCHOMIENIE APARATU I MODELU AI ---
const initMediaPipeAndCamera = async () => {
  try {
    if (!videoRef.value || !canvasContainerRef.value) return;

    const stream = await navigator.mediaDevices.getUserMedia({
      video: { width: { ideal: 1280 }, height: { ideal: 720 }, facingMode: 'user' },
      audio: false
    });
    videoRef.value.srcObject = stream;

    await new Promise((resolve) => {
      videoRef.value!.onloadedmetadata = () => resolve(true);
    });

    await videoRef.value.play();

    initThree(canvasContainerRef.value, videoRef.value);

    statusMessage.value = 'Wyszukiwanie silnika AI...';
    const vision = await FilesetResolver.forVisionTasks(
      'https://cdn.jsdelivr.net/npm/@mediapipe/tasks-vision@latest/wasm'
    );

    imageSegmenter = await ImageSegmenter.createFromOptions(vision, {
      baseOptions: {
        modelAssetPath: 'https://storage.googleapis.com/mediapipe-models/image_segmenter/selfie_segmenter/float16/latest/selfie_segmenter.tflite',
        delegate: 'GPU'
      },
      runningMode: 'VIDEO',
      outputCategoryMask: true,
      outputConfidenceMasks: false
    });

    statusMessage.value = 'Wirtualne tło aktywne!';
    isReady.value = true;

    processVideoFrame();
  } catch (error) {
    console.error(error);
    statusMessage.value = 'Błąd uruchamiania kamery lub modelu AI.';
  }
};

// --- GŁÓWNA PĘTLA OPERACYJNA ---
const processVideoFrame = () => {
  if (!videoRef.value || !imageSegmenter) return;

  const video = videoRef.value;

  if (video.currentTime !== lastVideoTime && video.readyState >= 2) {
    lastVideoTime = video.currentTime;
    const timestamp = performance.now();

    imageSegmenter.segmentForVideo(video, timestamp, (result) => {
      const mask = Array.isArray(result.categoryMask) ? result.categoryMask[0] : result.categoryMask;

      if (mask && maskCtx) {
        const width = mask.width;
        const height = mask.height;

        if (maskCanvas.width !== width || maskCanvas.height !== height) {
          maskCanvas.width = width;
          maskCanvas.height = height;
        }

        const maskData = mask.getAsFloat32Array() || mask.getAsUint8Array();

        if (maskData) {
          const imageData = maskCtx.createImageData(width, height);

          for (let i = 0; i < maskData.length; i++) {
            // Zamiast ostrego odcinania, mapujemy pełne spektrum (0.0 - 1.0) na pełną rozpiętość szarości (0 - 255).
            // Dzięki temu krawędzie zyskują naturalny antyaliasing.
            const intensity = Math.floor(maskData[i] * 255);
            const pixelIndex = i * 4;

            imageData.data[pixelIndex] = intensity;     // R
            imageData.data[pixelIndex + 1] = intensity; // G
            imageData.data[pixelIndex + 2] = intensity; // B
            imageData.data[pixelIndex + 3] = 255;       // A
          }

          maskCtx.putImageData(imageData, 0, 0);

          if (maskTexture) {
            maskTexture.needsUpdate = true;
          }
        }
      }
    });
  }

  if (renderer && scene && camera) {
    renderer.render(scene, camera);
  }

  animationFrameId = requestAnimationFrame(processVideoFrame);
};

const handleResize = () => {
  if (!canvasContainerRef.value || !renderer || !shaderMaterial) return;
  const width = canvasContainerRef.value.clientWidth;
  const height = canvasContainerRef.value.clientHeight;
  renderer.setSize(width, height);
};

// --- DEMONTAŻ KOMPONENTU ---
onMounted(() => {
  initMediaPipeAndCamera();
  window.addEventListener('resize', handleResize);
});

onBeforeUnmount(() => {
  if (animationFrameId) cancelAnimationFrame(animationFrameId);
  window.removeEventListener('resize', handleResize);

  if (renderer) {
    renderer.dispose();
    renderer.domElement.remove();
  }
  if (shaderMaterial) shaderMaterial.dispose();
  if (webcamTexture) webcamTexture.dispose();
  if (maskTexture) maskTexture.dispose();
  if (backgroundTexture) backgroundTexture.dispose();
  if (imageSegmenter) imageSegmenter.close();

  if (videoRef.value && videoRef.value.srcObject) {
    const stream = videoRef.value.srcObject as MediaStream;
    stream.getTracks().forEach(track => track.stop());
  }
});
</script>
